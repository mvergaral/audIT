% ============================================================================
%  Formulario T-12. Matriz de cumplimiento técnico y trazabilidad (FEP01 p. 62)
%  Subdocumento 3 según el Formulario T-21.
%
%  Une las columnas del T-12 (ID, descripción, cumple, componente, sección)
%  con los campos que el numeral 17.1 del Caso (p. 38) exige al catálogo:
%    funcional     identificador, descripción, actor, precondición,
%                  resultado esperado, prioridad y origen
%    no funcional  categoría, umbral numérico, método de verificación, y si
%                  es exigible a la compañía o a terceros
%
%  \requisitoF{id=RF-001, descripcion=..., actor=..., precondicion=...,
%              resultado=..., prioridad=Alta, origen=..., cumple=Sí,
%              componente=...}
%  \requisitoNF{id=RT-09.01, descripcion=..., categoria=Desempeño,
%               umbral=..., metodo=..., exigible=Compañía, prioridad=...,
%               origen=..., cumple=Sí, componente=...}
%
%  La columna «Sección de la propuesta» se completa sola con las secciones
%  donde el texto usa \req{ID} o \reqMargen{ID}. Si ningún párrafo atiende el
%  requisito, lo dice. Cada código queda como destino de enlace.
% ============================================================================

\DefinirFormulario{T-12}{Matriz de cumplimiento técnico y trazabilidad}{62}
  {El Proponente deberá declarar el cumplimiento de cada requerimiento de las
   Bases Técnicas del caso y de los requisitos transversales del Capítulo 3,
   indicando dónde se acredita en la propuesta. Cada requerimiento indica
   además los campos del catálogo exigidos por el numeral 17.1 del Caso
   (p. 38).}

\newcommand\requisitoF[1]{\RegistroNuevo{t12}{tipo=F,#1}}
\newcommand\requisitoNF[1]{\RegistroNuevo{t12}{tipo=NF,#1}}
